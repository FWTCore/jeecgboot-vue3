### 2.2 API 文件规范
```typescript
import { defHttp } from '/@/utils/http/axios';
import { Modal } from 'ant-design-vue';

enum Api {
  xxxList = '/xxx/list',
  xxxSave = '/xxx/add',
  xxxEdit = '/xxx/edit',
  xxxDelete = '/xxx/delete',
  batchXxxDelete = '/xxx/deleteBatch',
}

// 列表接口
export const xxxList = (params) => defHttp.get({ url: Api.xxxList, params });

// 保存或更新
export const saveOrUpdateXxx = (params, isUpdate) => {
  const url = isUpdate ? Api.xxxEdit : Api.xxxSave;
  return defHttp.post({ url: url, params });
};

// 删除接口
export const deleteXxx = (params, handleSuccess) => {
  return defHttp.delete({ url: Api.xxxDelete, params }, { joinParamsToUrl: true }).then(() => {
    handleSuccess();
  });
};

// 批量删除
export const batchXxxDelete = (params, handleSuccess) => {
  Modal.confirm({
    title: '确认删除',
    content: '是否删除选中数据',
    okText: '确认',
    cancelText: '取消',
    onOk: () => {
      return defHttp.delete({ url: Api.batchXxxDelete, data: params }, { joinParamsToUrl: true }).then(() => {
        handleSuccess();
      });
    },
  });
};
```

### 4.2 API 接口映射
| 功能 | 前端方法 | 后端接口 |
|------|---------|---------|
| 列表查询 | `overtimeList` | GET `/overtime/record/list` |
| 新增申请 | `saveOrUpdateOvertime` | POST `/overtime/record/add` |
| 编辑申请 | `saveOrUpdateOvertime` | PUT `/overtime/record/edit` |
| 删除 | `deleteOvertime` | DELETE `/overtime/record/delete` |
| 批量删除 | `batchDeleteOvertime` | DELETE `/overtime/record/deleteBatch` |
| 确认 | `confirmOvertime` | POST `/overtime/record/confirm` |
| 查询项目进度 | `queryProjectSchedule` | GET `/overtime/record/queryProjectSchedule` |
| 查询项目列表 | `getAllProject` | GET `/project/queryall` |

### 4.3 权限控制
- 普通用户：只能查看/编辑/删除自己的待确认记录
- `office:management` 权限用户：可查看所有记录、可确认记录

### 4.4 业务校验（后端实现，前端提示）
- 加班时长必须 > 0
- 加班时长最多一位小数
- 单次加班时长 ≤ 8 小时
- 已确认的记录不可编辑/删除

## 六、关键代码差异点

### 6.1 与 ScheduleLog 的差异
| 功能 | ScheduleLog | OvertimeRecord |
|------|-------------|----------------|
| 日期字段 | createTime (服务时间) | overtimeDate (加班日期) |
| 时长字段 | workHours (工时-天) | overtimeHours (加班时长-小时) |
| 补录功能 | 有 (remedy) | 无 |
| 确认功能 | 无 | 有 (confirmStatus) |
| 权限控制 | 无 | 有 (office:management) |
| 项目必填 | 是 | 否 |

### 6.2 确认功能实现
```typescript
// 操作栏新增确认按钮
function getTableAction(record) {
  const actions = [
    {
      label: '编辑',
      onClick: handleEdit.bind(null, record),
      ifShow: () => record.confirmStatus === 0, // 仅待确认可编辑
    },
    {
      label: '详情',
      onClick: handleDetail.bind(null, record),
    },
    {
      label: '删除',
      popConfirm: {
        title: '确定删除吗?',
        confirm: handleDelete.bind(null, record),
      },
      ifShow: () => record.confirmStatus === 0, // 仅待确认可删除
    },
  ];

  // 确认按钮（需要权限）
  if (hasPermission('office:management')) {
    actions.push({
      label: '确认',
      popConfirm: {
        title: '确定确认吗?',
        confirm: handleConfirm.bind(null, record),
      },
      ifShow: () => record.confirmStatus === 0,
    });
  }

  return actions;
}
```
